# Guard equivalence corpus (repo#193).
#
# One command per line. Blank lines and #-comments are skipped. Each case is
# sent to BOTH the canonical guard and Loom's vendored copy, and their
# permissionDecision values are compared under the "never weaker" rule (see
# test-guard-equivalence.sh).
#
# This file is DATA, deliberately. Several cases contain literals the guard
# itself denies (rm -rf /, DROP TABLE, cloud deletion verbs), so building this
# corpus inside a shell heredoc trips the guard on its own test fixtures. Add
# cases by editing this file, never by generating them from a shell command.
#
# Lines are commands only — no cwd. The harness runs every case from a scratch
# git repo it builds itself, so results do not depend on the developer's
# working directory or on any ambient worktree.

# ── quoting asymmetry (repo#197) ──────────────────────────────────────────
# Quoting an OPERATIVE argument used to defeat the literal catastrophic
# patterns outright — the guard enforced a spelling, not a policy. Canonical
# now dequotes inert spans AFTER sink-aware redaction, so these deny; the
# vendored copy still allows/asks, which is why they are declared divergences.
rm -rf "/"
rm -rf '/'
git push --force origin "main"
git push -f origin 'main'

# ── dequoting must NOT make read-only commands look dangerous (repo#197) ──
# This is the risk the dequote step introduces: a search or display command
# whose quoted ARGUMENT contains a destructive phrase. It is data to that
# command, and it must stay allowed. A regression here would deny ordinary
# code search in any repo that documents destructive commands — including
# this one.
grep -rn "rm -rf /" .
grep -F 'git push --force origin main' hooks/
rg --fixed-strings "DROP TABLE users" .
awk '/rm -rf \//' file.txt
printf '%s\n' "rm -rf /"
echo 'git push --force origin main'

# ── a shell reached through xargs/parallel (repo#429) ─────────────────────
# echo/printf piped into xargs/parallel, which then spawns a shell over the
# piped data. Both guards deny these: the vendored copy has no data-sink
# redaction at all (it denies any echo/printf of dangerous-looking text
# outright, see the DECLARED_WEAKER rows above), and canonical now recognises
# xargs/parallel as a shell-spawning segment so its own data-sink redaction is
# skipped and the raw scan still sees the payload.
echo 'rm -rf /' | xargs -I{} sh -c '{}'
echo 'rm -rf /' | xargs -0 bash -c '{}'
echo 'rm -rf /' | parallel sh -c '{}'

# ── ordinary xargs pipelines with no dangerous text must stay allowed ─────
echo hello | xargs -I{} echo {}
find . -name '*.tmp' | xargs rm

# ── query commands: the PATTERN is data, not a command (repo#311) ─────────
# jq/grep/egrep/fgrep/rg/inert-sed/awk MATCH AGAINST their pattern argument;
# they never execute it. These shapes — auditing this repo's own guard-decision
# log for the text of a catastrophic pattern — were denied purely because the
# pattern text appeared in the command string. They are the same false-positive
# class as the echo/printf rows above, and the vendored copy still denies them,
# so each has a DECLARED_WEAKER entry in test-guard-equivalence.sh.
#
# The shapes below deliberately carry a pipe or a non-fastpath command word:
# `grep`/`rg`/`jq` with no shell metacharacter at all are already admitted by
# the read-only fast path (see the section above), so a corpus case without one
# would prove nothing about the redaction under test.
jq -r 'select(.pattern == "catastrophic:git push --force origin main") | .ts' log.jsonl
jq -r 'select(.command | test("rm -rf /"))' log.jsonl
egrep 'rm -rf /' log.jsonl
fgrep 'git push --force origin main' log.jsonl
grep -c 'rm -rf /' log.jsonl | wc -l
rg 'git push --force origin main' . | head -3
sed -n 's|rm -rf /|X|p' log.jsonl
awk '$0 ~ "git push --force origin main" {print}' log.jsonl

# ── query-sink EXCLUSIONS and safety floor (repo#311) ─────────────────────
# Sub-forms that ACT rather than query are vetoed out of sink treatment, and
# the shared safety floor (command substitution, pipe-to-shell) is unchanged.
# Every row here must still DENY in BOTH guards — a weaker verdict on any of
# them is a real regression, not a declarable divergence.
sed -i 's|x|rm -rf /|' f.txt
sed -n 's|x|y|w rm -rf /' f.txt
sed -n 'e rm -rf /' f.txt
awk '{system("rm -rf /")}' f.txt
awk '{print | "rm -rf /"}' f.txt
rg --pre 'rm -rf /' . | head -3
jq "$(printf 'rm -rf /')" f.json
grep 'rm -rf /' f.txt | sh
